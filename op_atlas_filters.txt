To make the filters work with the responsive grid, the key is:

Filter the cards themselves, not the grid.
Keep the grid container visible.
Hide non-matching cards with display:none.
Let CSS Grid automatically reflow the remaining cards.

Here's the complete filtering logic that works perfectly with a .opportunity-grid.

Better Filter Function
const cards =
document.querySelectorAll(".card");

function filterOpportunities(){

    const search =
        document
        .getElementById("search")
        .value
        .toLowerCase()
        .trim();

    const cause =
        document
        .getElementById("cause")
        .value;

    const skill =
        document
        .getElementById("skill")
        .value;

    const contribution =
        document
        .getElementById("contribution")
        .value;

    let visibleCount = 0;

    cards.forEach(card => {

        const title =
            card.dataset.title.toLowerCase();

        const cardCause =
            card.dataset.cause;

        const cardSkills =
            card.dataset.skills;

        const cardContribution =
            card.dataset.contribution;

        const searchableText = (
            title +
            " " +
            cardCause +
            " " +
            cardSkills +
            " " +
            cardContribution
        ).toLowerCase();

        const searchMatch =
            search === "" ||
            searchableText.includes(search);

        const causeMatch =
            cause === "All" ||
            cardCause === cause;

        const skillMatch =
            skill === "All" ||
            cardSkills.includes(skill);

        const contributionMatch =
            contribution === "All" ||
            cardContribution.includes(contribution);

        const visible =
            searchMatch &&
            causeMatch &&
            skillMatch &&
            contributionMatch;

        if(visible){

            card.style.display = "";

            visibleCount++;

        }
        else{

            card.style.display = "none";

        }

    });

    document.getElementById(
        "resultsCount"
    ).textContent =
        `${visibleCount} opportunities found`;

}

Connect All Filters
[
    "search",
    "cause",
    "skill",
    "contribution"
].forEach(id => {

    document
        .getElementById(id)
        .addEventListener(
            id === "search"
            ? "input"
            : "change",
            filterOpportunities
        );

});

Reset Button
function resetFilters(){

    document.getElementById("search").value = "";

    document.getElementById("cause").value = "All";

    document.getElementById("skill").value = "All";

    document.getElementById("contribution").value = "All";

    filterOpportunities();

}

<button onclick="resetFilters()">
    Reset Filters
</button>
