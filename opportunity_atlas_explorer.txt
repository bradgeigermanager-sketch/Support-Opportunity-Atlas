<script>

const opportunities =
document.querySelectorAll(".card");

const searchInput =
document.getElementById("search");

const causeFilter =
document.getElementById("cause");

const skillFilter =
document.getElementById("skill");

const contributionFilter =
document.getElementById("contribution");

function filterOpportunities(){

    const search =
    searchInput.value.toLowerCase();

    const cause =
    causeFilter.value;

    const skill =
    skillFilter.value;

    const contribution =
    contributionFilter.value;

    let visibleCount = 0;

    opportunities.forEach(card => {

        const title =
        card.dataset.title.toLowerCase();

        const causeValue =
        card.dataset.cause;

        const skills =
        card.dataset.skills;

        const contributionValue =
        card.dataset.contribution;

        const searchMatch =
        title.includes(search) ||
        causeValue.toLowerCase().includes(search) ||
        skills.toLowerCase().includes(search);

        const causeMatch =
        cause === "All" ||
        causeValue === cause;

        const skillMatch =
        skill === "All" ||
        skills.includes(skill);

        const contributionMatch =
        contribution === "All" ||
        contributionValue.includes(contribution);

        if(
            searchMatch &&
            causeMatch &&
            skillMatch &&
            contributionMatch
        ){
            card.style.display = "block";
            visibleCount++;
        }
        else{
            card.style.display = "none";
        }

    });

    document.getElementById(
        "resultsCount"
    ).innerHTML =
    visibleCount +
    " opportunity(s) found";

}

function resetFilters(){

    searchInput.value="";

    causeFilter.value="All";

    skillFilter.value="All";

    contributionFilter.value="All";

    filterOpportunities();

}

searchInput.addEventListener(
    "input",
    filterOpportunities
);

causeFilter.addEventListener(
    "change",
    filterOpportunities
);

skillFilter.addEventListener(
    "change",
    filterOpportunities
);

contributionFilter.addEventListener(
    "change",
    filterOpportunities
);

filterOpportunities();

</script>
