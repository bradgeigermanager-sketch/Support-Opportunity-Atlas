For the Opportunity Atlas, I'd implement responsive numbered pagination that adapts to screen size:

🖥️ Desktop: 1 2 3 4 5 ... 20
📱 Mobile: 1 ... 4 5 6 ... 20
Very small screens: only the current page plus neighbors.

<div class="pagination-container">

    <button id="prevPage">
        ← Previous
    </button>

    <div id="pageNumbers"></div>

    <button id="nextPage">

Smart Responsive Pagination Logic

Replace your existing page-number code with:

function buildPageNumbers(){

    const container =
        document.getElementById(
            "pageNumbers"
        );

    container.innerHTML = "";

    const totalPages =
        Math.max(
            1,
            Math.ceil(
                filteredCards.length /
                cardsPerPage
            )
        );

    const mobile =
        window.innerWidth < 768;

    let pages = [];

    if(totalPages <= 7){

        for(let i=1;i<=totalPages;i++){

            pages.push(i);

        }

    }else{

        if(mobile){

            pages.push(1);

            if(currentPage > 3){

                pages.push("...");

            }

            const start =
                Math.max(
                    2,
                    currentPage - 1
                );

            const end =
                Math.min(
                    totalPages - 1,
                    currentPage + 1
                );

            for(
                let i=start;
                i<=end;
                i++
            ){
                pages.push(i);
            }

            if(
                currentPage <
                totalPages - 2
            ){
                pages.push("...");
            }

            pages.push(totalPages);

        }else{

            pages.push(1);

            const start =
                Math.max(
                    2,
                    currentPage - 2
                );

            const end =
                Math.min(
                    totalPages - 1,
                    currentPage + 2
                );

            if(start > 2){

                pages.push("...");

            }

            for(
                let i=start;
                i<=end;
                i++
            ){
                pages.push(i);
            }

            if(
                end <
                totalPages - 1
            ){
                pages.push("...");
            }

            pages.push(totalPages);

        }

    }

    pages.forEach(page => {

        if(page === "..."){

            const span =
                document.createElement(
                    "span"
                );

            span.className =
                "page-btn ellipsis";

            span.textContent =
                "...";

            container.appendChild(
                span
            );

            return;

        }

        const btn =
            document.createElement(
                "button"
            );

        btn.textContent =
            page;

        btn.className =
            "page-btn";

        if(page === currentPage){

            btn.classList.add(
                "active"
            );

        }

        btn.addEventListener(
            "click",
            () => {

                currentPage = page;

                renderPage();

            }
        );

        container.appendChild(
            btn
        );

    });

}

Update Render Function

At the end of renderPage():

buildPageNumbers();


And whenever the browser is resized:

window.addEventListener(
    "resize",
    buildPageNumbers
);
